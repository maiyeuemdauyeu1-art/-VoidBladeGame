import os import time import subprocess from google.cloud import compute_v1 from google.oauth2 import service_account # ====== CẤU HÌNH ====== PROJECT_ID = "your-project-id" # Thay bằng Project ID của bạn ZONE = "us-central1-a" # Vùng máy chủ INSTANCE_NAME = "android-cloud-phone" CREDENTIALS_PATH = "path-to-service-account-key.json" # File key JSON # ====== KHỞI TẠO ====== credentials = service_account.Credentials.from_service_account_file(CREDENTIALS_PATH) instances_client = compute_v1.InstancesClient(credentials=credentials) # ====== 1. TẠO VM CHẠY ANDROID ====== def create_android_vm(): print("📱 Đang tạo máy ảo Android trên đám mây...") # Cấu hình VM instance = compute_v1.Instance() instance.name = INSTANCE_NAME instance.machine_type = f"zones/{ZONE}/machineTypes/n2-standard-4" # 4 CPU, 16GB RAM instance.zone = ZONE # Boot disk - dùng Android 11 image (có sẵn trên GCP) disk = compute_v1.AttachedDisk() disk.boot = True disk.auto_delete = True initialize_params = compute_v1.AttachedDiskInitializeParams() initialize_params.source_image = "projects/android-os-ember/global/images/android-11-20230101" # Android 11 initialize_params.disk_size_gb = 30 disk.initialize_params = initialize_params instance.disks = [disk] # Network network_interface = compute_v1.NetworkInterface() network_interface.name = "global/networks/default" instance.network_interfaces = [network_interface] # Mở port ADB (5555) và VNC (5900) firewall_rule = compute_v1.Firewall() firewall_rule.name = "allow-adb-vnc" firewall_rule.allowed = [ compute_v1.Allowed(IP_protocol="tcp", ports=["5555", "5900"]) ] firewall_rule.source_ranges = ["0.0.0.0/0"] # Tạo VM operation = instances_client.insert(project=PROJECT_ID, zone=ZONE, instance_resource=instance) operation.result() # Đợi tạo xong # Lấy IP công khai instance = instances_client.get(project=PROJECT_ID, zone=ZONE, instance=INSTANCE_NAME) external_ip = instance.network_interfaces[0].access_configs[0].nat_i_p print(f"✅ Đã tạo xong! IP: {external_ip}") return external_ip # ====== 2. KẾT NỐI ADB ====== def connect_adb(ip): print("🔌 Kết nối ADB...") subprocess.run(f"adb connect {ip}:5555", shell=True) time.sleep(3) subprocess.run("adb devices", shell=True) # ====== 3. CÀI ĐẶT APP (APK) ====== def install_apk(apk_path): print("📦 Cài đặt APK...") subprocess.run(f"adb install {apk_path}", shell=True) # ====== 4. ĐIỀU KHIỂN TỪ XA ====== def control_phone(ip): print(""" ======================================== 🎮 ĐIỀU KHIỂN ĐIỆN THOẠI ĐÁM MÂY: - Kết nối VNC: dùng client bất kỳ vào {ip}:5900 - Hoặc dùng scrcpy: scrcpy --serial {ip}:5555 - ADB commands: adb shell ======================================== """) # ====== 5. TẢI APK VÀ CÀI ====== def download_and_install_apk(url): print("⬇️ Tải APK từ URL...") apk_name = url.split("/")[-1] subprocess.run(f"wget -O {apk_name} {url}", shell=True) install_apk(apk_name) os.remove(apk_name) # ====== MAIN ====== if __name__ == "__main__": # Tạo máy ảo ip = create_android_vm() # Kết nối ADB connect_adb(ip) # Tải và cài APK (ví dụ cài Chrome) download_and_install_apk("https://apkpure.com/chrome/com.android.chrome/download") # Hướng dẫn điều khiển control_phone(ip) # Mở shell để dùng trực tiếp print("🚀 Mở shell Android...") subprocess.run(f"adb -s {ip}:5555 shell", shell=True)
